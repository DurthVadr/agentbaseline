{{/* OpenGraph and Twitter cards, shared by both pages. These lived in layouts/index.html only, so /controls — which is the citation target, the URL someone shares when they are pointing at a specific control — previewed as nothing at all. Same failure as the event strip: markup in one layout, silently absent from the other. Called with a dict so each page supplies its own title and description, and optionally its own card: {{ partial "social-meta.html" (dict "ctx" . "title" "…" "desc" "…" "image" "…") }} */}} {{- $t := .title | default "Agent Baseline" -}} {{- $d := .desc -}} {{- $u := .ctx.Permalink | default .ctx.Site.BaseURL -}} {{- /* trailingSlash is false in vercel.json, so /controls/ 308s to /controls. Declaring the redirecting form as canonical hands two readers two different URLs for the same page. The root keeps its slash; everything else loses it. */ -}} {{- if ne $u .ctx.Site.BaseURL -}}{{- $u = strings.TrimSuffix "/" $u -}}{{- end -}} {{- /* The card carries the control count, and platforms cache og:image by URL — so fixing the bytes in place leaves the old number in every preview that has already been fetched. Putting the count in the filename means the URL changes exactly when the number does, which is the only thing on the card that has ever been wrong. */ -}} {{- /* Versioning the filename by count defeats the platforms' og:image cache, but only bin/build-social writes that file and it never runs in CI — it needs Chrome on a Mac. So the count can move without the image existing, and an og:image that 404s is worse than one showing a stale number: the preview renders blank rather than slightly wrong. Fall back to the unversioned card, which is always present. */ -}} {{- $n := len hugo.Data.controls.controls -}} {{- $file := printf "social-card-%d.png" $n -}} {{- if not (fileExists (printf "static/%s" $file)) -}} {{- warnf "social-card-%d.png is missing — og:image falls back to social-card.png, which may show a stale count. Run bin/build-social." $n -}} {{- $file = "social-card.png" -}} {{- end -}} {{- /* A page with a card of its own passes "image" (a file in static/), e.g. the second paper's page, whose share should name that paper. Missing, it falls back to the baseline's card rather than an og:image that 404s. */ -}} {{- with .image -}} {{- if fileExists (printf "static/%s" .) -}} {{- $file = . -}} {{- else -}} {{- warnf "%s is missing — og:image falls back to %s. Run bin/build-social --paper." . $file -}} {{- end -}} {{- end -}} {{- $card := printf "%s%s" .ctx.Site.BaseURL $file -}}
